package sse_client

import "time"


// 定义通用接口
type SSEResponse interface {
	GetEventType() string
	GetCommonFields() CommonFields
}

type CommonFields struct {
	LogID     string    `json:"log_id"`
	TreeID    string    `json:"tree_id"`
	QAID      string    `json:"qa_id"`
	PkgIndex  int       `json:"pkg_index"`
	CreatedAt time.Time `json:"created_at"`
	Done      bool      `json:"done"`
}

// 具体实现
type TextResponse struct {
	CommonFields
	NodeData *NodeData `json:"node_data"`
}

func (t *TextResponse) GetEventType() string          { return TextEvent }
func (t *TextResponse) GetCommonFields() CommonFields { return t.CommonFields }

type ErrorResponse struct {
	CommonFields
	Error *Error `json:"error"`
}

func (e *ErrorResponse) GetEventType() string          { return ErrorEvent }
func (e *ErrorResponse) GetCommonFields() CommonFields { return e.CommonFields }

type AskBackResponse struct {
	CommonFields
	AskBack []string `json:"ask_back"`
	NodeID  string   `json:"node_id"`
}

func (a *AskBackResponse) GetEventType() string          { return AskBackEvent }
func (a *AskBackResponse) GetCommonFields() CommonFields { return a.CommonFields }

type TitleResponse struct {
	CommonFields
	Title string `json:"title"`
}

func (t *TitleResponse) GetEventType() string          { return TitleEvent }
func (t *TitleResponse) GetCommonFields() CommonFields { return t.CommonFields }

type CloseResponse struct {
	CommonFields
	Reason string `json:"reason"`
}

func (c *CloseResponse) GetEventType() string          { return DoneEvent }
func (c *CloseResponse) GetCommonFields() CommonFields { return c.CommonFields }

type Error struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
}

type NodeData struct {
	ID      string `json:"id"`
	Pos     int    `json:"pos"`
	Content string `json:"content"`
	Finish  bool   `json:"finish"` // 当前节点是否结束
}

const (
	TextEvent    = "text"
	ErrorEvent   = "error"
	AskBackEvent = "ask_back"
	TitleEvent   = "title"
	DoneEvent    = "done"
)
